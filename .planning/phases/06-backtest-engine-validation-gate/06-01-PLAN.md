    ```python
    from __future__ import annotations
    from datetime import date, datetime, timezone
    from typing import Iterator
    import pandas as pd
    from sqlalchemy import text
    from app.flows._base import sync_session

    def iter_business_days(
        start: date,
        end: date,
        exclude_date_range: tuple[date, date] | None = None,
    ) -> Iterator[date]:
        """Yield business days in [start, end] inclusive. Skip an optional inclusive exclusion range.

        Uses pandas bdate_range freq='B'. Acceptable inaccuracy: includes non-NYSE holidays like
        Columbus Day; the replay loop tolerates dates with no price data (skips them).
        """
        rng = pd.bdate_range(start=start, end=end, freq="B")
        ex_start = exclude_date_range[0] if exclude_date_range else None
        ex_end = exclude_date_range[1] if exclude_date_range else None
        for ts in rng:
            d = ts.date()
            if ex_start is not None and ex_start <= d <= ex_end:
                continue
            yield d

    def run_backtest(
        start: date,
        end: date,
        *,
        slice_type: str = "main",
        exclude_date_range: tuple[date, date] | None = None,
        override_gate: bool | None = None,
    ) -> dict:
        """Skeleton entrypoint. Full replay body added in plan 04.

        Currently iterates as_of dates and proves sync_session opens cleanly. Returns a
        summary dict with no real fills or stats.
        """
        dates_iterated = 0
        as_of_last: date | None = None
        with sync_session() as session:
            # Touch DB to prove session works; trivial SELECT 1
            session.execute(text("SELECT 1")).scalar()
            for as_of in iter_business_days(start, end, exclude_date_range=exclude_date_range):
                dates_iterated += 1
                as_of_last = as_of
        return {
            "run_id": None,
            "slice_type": slice_type,
            "dates_iterated": dates_iterated,
            "as_of_last": as_of_last,
        }
    ```

    ```python
    from __future__ import annotations
    from config import CONFIG  # repo-root config.py via sys.path augmented in __init__.py

    def simulate_fill(
        price: float,
        side: str,
        qty: float,
        *,
        transaction_cost_bps: float | None = None,
    ) -> dict:
        """Deterministic simulated fill. No Alpaca calls.

        side must be 'buy' or 'sell'. Applies transaction_cost_bps as a symmetric haircut:
        buy fills at price * (1 + bps/10000), sell at price * (1 - bps/10000).
        """
        if side not in ("buy", "sell"):
            raise ValueError(f"side must be 'buy' or 'sell', got {side!r}")
        bps = transaction_cost_bps if transaction_cost_bps is not None else float(CONFIG.risk.transaction_cost_bps)
        adj = bps / 10000.0
        if side == "buy":
            fill_price = price * (1.0 + adj)
        else:
            fill_price = price * (1.0 - adj)
        return {"fill_price": float(fill_price), "cost_bps": float(bps), "qty": float(qty)}
    ```

    ```python
    """FR-6.1: a row with ingestion_timestamp > as_of must NOT appear in any backtest query."""
    from datetime import timedelta
    from sqlalchemy import text
    from tests.conftest import requires_db

    @requires_db
    def test_future_row_rejected_by_as_of_filter(sync_engine, backtest_test_as_of):
        as_of = backtest_test_as_of
        future_ts = as_of + timedelta(days=1)
        sym = "TEST_FUTURE_BT"
        try:
            with sync_engine.begin() as conn:
                conn.execute(
                    text(
                        "INSERT INTO price_bars (time, symbol, open, high, low, close, volume, ingestion_timestamp) "
                        "VALUES (:t, :sym, :px, :px, :px, :px, 0, :ingest)"
                    ),
                    {"t": as_of, "sym": sym, "px": 999.99, "ingest": future_ts},
                )
            with sync_engine.connect() as conn:
                row = conn.execute(
                    text(
                        "SELECT close FROM price_bars "
                        "WHERE symbol = :sym AND ingestion_timestamp <= :as_of"
                    ),
                    {"sym": sym, "as_of": as_of},
                ).fetchone()
            assert row is None, f"FR-6.1 violation: future-timestamped row leaked through as_of filter: {row}"
        finally:
            with sync_engine.begin() as conn:
                conn.execute(text("DELETE FROM price_bars WHERE symbol = :sym"), {"sym": sym})
    ```
