**Step 6: Create frontend/src/hooks/useSSE.ts.** Mark with `'use client'`:
```typescript
'use client';
import { useContext, useEffect, useState } from 'react';
import { SSEContext } from '@/context/SSEContext';

export function useSSE(channel: 'signals' | 'positions' | 'rl_state' | 'alerts'): {
  lastEvent: unknown;
  connected: boolean;
} {
  const ctx = useContext(SSEContext);
  const [lastEvent, setLastEvent] = useState<unknown>(null);
  useEffect(() => {
    ctx.addListener(channel, setLastEvent);
    return () => ctx.removeListener(channel, setLastEvent);
  }, [channel, ctx]);
  return { lastEvent, connected: ctx.connected };
}
```
